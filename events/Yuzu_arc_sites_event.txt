					spawn_system = {
						min_distance = 20
						max_distance = 30

					}
					if = {
						limit = { last_created_system = { NOT = { has_hyperlane_to = prev } } }

							from = this
							to = last_created_system
						}
					}
					else = {
						add_extra_hyperlane_to_spawned_system_effect = yes
					}
				}
				hidden_effect = {
					country_event = { id = Yuzu_window_event.1 days = 1 }

			else = {
				hidden_effect = {
					country_event = { id = Yuzu_window_event.2 days = 1 }
				}
			}
		}

fleet_event = {
	id = Yuzu_arc_sites_event.26
	title = Yuzu_arc_sites_event.26.name
	desc = Yuzu_arc_sites_event.26.desc
	picture = GFX_evt_relic_world
	show_sound = event_default
	location = from

	archaeology = yes

	is_triggered_only = yes

	# 奖励：按玩家当前合金科技水平推进下一级（见 directions/directive76_easter_egg_system.md 3.5）
	option = {
		name = Yuzu_arc_sites_event.26.a
		owner = {
			if = {
				limit = { NOT = { has_technology = tech_alloys_1 } }
				add_research_option = tech_alloys_1
				add_tech_progress = { tech = tech_alloys_1 progress = 0.25 }
			}
			else_if = {
				limit = { NOT = { has_technology = tech_alloys_2 } }
				add_research_option = tech_alloys_2
				add_tech_progress = { tech = tech_alloys_2 progress = 0.25 }
			}
			else_if = {
				limit = { NOT = { has_technology = tech_fe_forge_1 } }
				add_research_option = tech_fe_forge_1
				add_tech_progress = { tech = tech_fe_forge_1 progress = 0.25 }
			}
			else_if = {
				limit = { NOT = { has_technology = tech_fe_forge_2 } }
				add_research_option = tech_fe_forge_2
				add_tech_progress = { tech = tech_fe_forge_2 progress = 0.25 }
			}
			else = {
				add_monthly_resource_mult = {
					resource = engineering_research
					value = 12
					min = @tier1researchmin
					max = @tier1researchmax
				}
				add_monthly_resource_mult = {
					resource = alloys
					value = 12
					min = @tier1materialmin
					max = 100000
				}
			}
		}
	}
}
