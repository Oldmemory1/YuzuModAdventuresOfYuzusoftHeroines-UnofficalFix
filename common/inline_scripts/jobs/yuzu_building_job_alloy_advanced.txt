# Regular Empires - Normal (minerals -> alloys)
triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
			is_catalytic_empire = no
		}
	}
	modifier = {
		job_yuzu_smelter_add = $AMOUNT$
	}
}
