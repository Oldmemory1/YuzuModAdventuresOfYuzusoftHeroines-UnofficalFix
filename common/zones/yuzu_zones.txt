# Yuzu Arcology Zones
# These are Yuzu-specific copies of the vanilla arcology zones from 04_secondary_zones.txt.
# They use the yuzu_arcology zone set and swap to Yuzu-specific visual districts.
# Zone set: yuzu_arcology
# Slot:     slot_yuzu_arcology_urban_01/02/03
# District: district_yuzu_city_urban_1/2/3
#
# 每个 arcology zone 都带 planet_modifier = { zone_building_slots_add = 3 }，与 vanilla 的
# shared_*_zone inline_script 一致（vanilla 的 12 个 arcology zone 全部给 3 个建筑槽）。
# 少了这一条，塔楼区划里就放不下该 zone 家族的建筑（研究所 / 要塞 / 商业楼等）。

zone_yuzu_industrial_arcology = {

	icon = GFX_district_specialization_industrial

	base_buildtime = 360
	resources = {
		category = planet_zones
		cost = {
			minerals = 1000
		}
	}

	swap_type = district_yuzu_city_industry
	swap_type_weight = 5

	zone_sets = {
		yuzu_arcology
	}

	included_building_sets = {
		industrial
		automation
		origin
	}

	inline_script = {
		script = jobs/yuzu_zone_job_goods
		AMOUNT = 600
	}
	inline_script = {
		script = jobs/yuzu_zone_job_alloy
		AMOUNT = 600
	}

	planet_modifier = {
		zone_building_slots_add = 3
	}

	triggered_district_planet_modifier = {
		planet_housing_add = @zone_arcology_housing_negative
	}
}

# 医疗（人口增长）特化。写法对照工作坊 mod「小狐狸的底层框架」(3243731913) 的
# zone_restoration_arcology：zone 图标记用原版 GFX_district_specialization_healthcare，
# 建筑集 medical + entertainment，岗位走 job_healthcare（医务工作者，原版唯一带
# pop_growth / pop_assembly / amenities 的常规岗位）。AMOUNT 与其它 yuzu zone 同单位
# （1/100 个岗位），300 = 每级 3 个医务工作者。
zone_yuzu_healthcare_arcology = {
	icon = GFX_district_specialization_healthcare

	base_buildtime = 360
	resources = {
		category = planet_zones
		cost = {
			minerals = 1000
		}
	}

	# 医务工作者岗位与医疗建筑都只对非格式塔开放，格式塔帝国没必要看到这个 zone
	potential = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
	}

	swap_type = district_yuzu_city_healthcare
	swap_type_weight = 5

	zone_sets = {
		yuzu_arcology
	}

	included_building_sets = {
		medical
		entertainment
		automation
		origin
	}

	inline_script = {
		script = jobs/yuzu_zone_job_healthcare
		AMOUNT = 300
	}

	planet_modifier = {
		zone_building_slots_add = 3
	}

	triggered_district_planet_modifier = {
		planet_housing_add = @zone_arcology_housing_negative
