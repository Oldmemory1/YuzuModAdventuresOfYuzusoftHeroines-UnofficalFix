zone_yuzu_unity_arcology = {
	icon = GFX_district_specialization_unity

	base_buildtime = 360
	resources = {
		category = planet_zones
		cost = {
			minerals = 1000
		}
	}

	# Match vanilla shared_unity_zone: administrative center is for non-spiritualists
	potential = {
		hidden_trigger = { exists = owner }
		owner = {
			is_wilderness_empire = no
			is_spiritualist = no
			NOT = { has_valid_civic = civic_machine_servitor }
		}
	}

	swap_type = district_yuzu_city_administrative
	swap_type_weight = 5

	zone_sets = {
		yuzu_arcology
	}

	included_building_sets = {
		unity
		automation
		origin
	}

	inline_script = {
		script = jobs/yuzu_zone_job_unity
		AMOUNT = 1000
	}

	convert_to = {
		zone_yuzu_unity_spiritualist_arcology
	}
}
