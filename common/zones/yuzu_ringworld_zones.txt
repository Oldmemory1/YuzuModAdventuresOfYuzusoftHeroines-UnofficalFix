# 柚子环世界 zone
#
# 参照 common/zones/yuzu_zones.txt 的城市 zone 写法：
# - 360 天、矿物 1000、无 zone 级维护；
# - 每个 zone 给 +3 建筑槽；
# - 每个 zone 给 +5000 住房（由 triggered_district_planet_modifier 提供，按区划等级叠加）；
# - 岗位通过现有 inline script 分发，AMOUNT 直接写字面量（1/100 个岗位，再乘区划等级）。
#
# zone 集：
#   yuzu_ring_world_urban      —— 两座柚子环世界共用：混合工业、重工业、民用工业、三种精炼、
#                                 贸易、物理学/社会学/工程学三种研究、凝聚力
#   yuzu_sunset_ring_world_dream  —— 夕阳环世界专属：发电、采矿、农业
#
# 槽位见 common/zone_slots/yuzu_ringworld_zone_slots.txt。
# 注意：混合工业按“每个特化合计 10000 岗位”拆成 goods 5000 + alloy 5000；
# 若本意是合金/消费品各 10000，直接把两个 AMOUNT 都改成 10000。

# =============================================================================
# 共享 urban zone（深渊环世界与夕阳环世界都能用）
# =============================================================================

zone_yuzu_ringworld_industrial = {
	icon = GFX_district_specialization_industrial

	base_buildtime = 360
	resources = {
		category = planet_zones
		cost = {
			minerals = 1000
		}
	}

	potential = {
		exists = owner
		OR = {
			is_planet_class = pc_yuzu_abyss_ringworld
			is_planet_class = pc_yuzu_sunset_ringworld
		}
	}

	swap_type = district_yuzu_ringworld_industrial
	swap_type_weight = 5

	zone_sets = {
		yuzu_ring_world_urban
	}

	included_building_sets = {
		industrial
		automation
		origin
	}

	inline_script = {
		script = jobs/yuzu_zone_job_elite_artisan
		AMOUNT = 5000
	}
	inline_script = {
		script = jobs/yuzu_zone_job_elite_metallurgist
		AMOUNT = 5000
	}

	planet_modifier = {
		zone_building_slots_add = 3
	}

	triggered_district_planet_modifier = {
		planet_housing_add = 5000
	}
}
