# YUZU 高级岗位：冶金 / 铸造类
#
# CSV（data_design.csv）中的 job_yuzu_* 是本地化与图标前缀；岗位定义 key 去掉 job_ 前缀：
#   job_yuzu_elite_foundry                      -> yuzu_elite_foundry
#   job_yuzu_elite_catalytic_technician         -> yuzu_elite_catalytic_technician（placeholder / swap 显示）
#   job_yuzu_elite_foundry_manager              -> yuzu_elite_foundry_manager
#   job_yuzu_elite_catalytic_technician_manager -> yuzu_elite_catalytic_technician_manager（placeholder / swap 显示）
#
# 催化分支沿用原版 foundry 的写法：同一个实际岗位，swappable_data 只负责换显示名/图标，
# 实际产出与维护费由 resources 内带 industrial_jobs_are_catalytic_trigger 的条件块决定。

#####################
# Specialist Jobs
#####################

yuzu_elite_foundry = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = building_foundry_1
			icon = yuzu_elite_metallurgist
		}
		# Catalytic Swap
		swap_type = {
			trigger = {
				industrial_jobs_are_catalytic_trigger = yes
			}
			name = yuzu_elite_catalytic_technician
			desc = job_yuzu_elite_catalytic_technician_desc
			icon = yuzu_elite_metallurgist
			building_icon = building_foundry_1
			weight = 5
		}
	}

	tags = { alloys }

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
	}

	resources = {
		category = planet_metallurgists
		# 非催化：矿物 -> 合金 + 工程学
		produces = {
			trigger = {
				industrial_jobs_are_catalytic_trigger = no
			}
			alloys = 12
			engineering_research = 4
		}
		upkeep = {
			trigger = {
				industrial_jobs_are_catalytic_trigger = no
			}
			minerals = 40
		}
		# 催化加工：食物 -> 合金 + 工程学
		produces = {
			trigger = {
				industrial_jobs_are_catalytic_trigger = yes
			}
			alloys = 15
			engineering_research = 5
		}
		upkeep = {
			trigger = {
				industrial_jobs_are_catalytic_trigger = yes
			}
			food = 60
		}
	}

	weight = {
		weight = @specialist_job_weight
		modifier = {
			factor = 2
			is_enslaved = yes
			can_take_servant_job = no
		}
		modifier = {
			factor = 5
			has_trait = trait_cyborg_creed_of_construction
		}
	}

	inline_script = "jobs/automodding_priority_alloys"
}

# 仅供 Catalytic Swap 显示与本地化使用；实际岗位仍是 yuzu_elite_foundry。
