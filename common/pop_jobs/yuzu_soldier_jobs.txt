	swappable_data = {

			condition_string = SOLDIER_JOB_TRIGGER
			building_icon = building_stronghold
			icon = yuzu_elite_soldier
		}
	}

	tags = { soldier }

	country_modifier = {
		country_naval_cap_add = 20
	}

	planet_modifier = {
		planet_stability_add = 1
		planet_defense_armies_add = 2
	}

	resources = {

		upkeep = {

		}
	}

		weight = @soldier_job_weight
	}
}
