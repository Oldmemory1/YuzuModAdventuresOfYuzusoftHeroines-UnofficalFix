# YUZU 精英研究岗位定义
# 数值来源：yuzu_advanced_jobs/data_design.csv 第 20-23 行
# 结构参考：原版 common/pop_jobs/02_specialist_jobs.txt（physicist）与 01_ruler_jobs.txt（politician）
# 共 4 个岗位：1 个高级研究员（specialist）+ 3 个学科研究主管（ruler）。
# CSV 中的 violate_motes 为拼写错误，实际资源 key 为 volatile_motes。
# CSV 中社会学主管写的是 planet_sociologists_produces_mult / planet_sociologists_upkeep_mult，
# 但原版 4.5 没有 planet_sociologists 经济类别；社会学研究岗位类别实际为 planet_biologists，
# 因此这里映射为 planet_biologists_produces_mult / planet_biologists_upkeep_mult。

yuzu_elite_researcher = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = building_research_lab_1
			icon = yuzu_elite_researcher
		}
	}

	tags = { research }

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		hidden_trigger = { exists = owner }
		complex_specialist_job_check_trigger = yes
		owner = { is_gestalt = no }
	}

	resources = {
		category = planet_researchers
		produces = {
			physics_research = 6
			society_research = 6
			engineering_research = 6
			unity = 5
		}
		upkeep = {
			consumer_goods = 12
		}
	}

	weight = {
		weight = @specialist_job_weight
	}

	inline_script = "jobs/automodding_priority_research"
}
