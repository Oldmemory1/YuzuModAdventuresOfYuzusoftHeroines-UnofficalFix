add_yuzu_leader_trait_yuzu = {
	if = {
		limit = {
			species = {
				has_trait = trait_yuzu
			}
			NOT = { has_trait = leader_trait_immortal_yuzu }
		}
		add_trait = leader_trait_yuzu
	}
}

remove_yuzu_leader_trait_yuzu = {
	if = {
		limit = {
			has_trait = leader_trait_yuzu
		}
		remove_trait = leader_trait_yuzu
	}
}

set_yuzu_leader_skill_single = {
	#split
	# 只在记录值>=2时才做记忆传递：把(该类历史最大-1)存入国家变量 yuzu_leader_skill，下限2
	if = {
		limit = {
			owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value >= 2 } }
		}
		owner = {
			set_variable = { which = yuzu_leader_skill value = yuzu_$LEADER_CLASS$_skill }
			change_variable = { which = yuzu_leader_skill value = -1 }
		}
		if = {
			limit = {
				owner = { check_variable = { which = yuzu_leader_skill value < 2 } }
			}
			owner = { set_variable = { which = yuzu_leader_skill value = 2 } }
		}
		# 用 add_skill 逐级抬到目标，跨过4/8时正常给出特化/命定选择
		while = {
			count = 8
			if = {
				limit = { has_base_skill < value:leader_yuzu_skill }
				add_skill = 1
			}
		}
	}
}

update_and_set_yuzu_leader_skill = {
	switch = {
		trigger = has_base_skill
		#split#
		10 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 10 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 10 } }
			}
		}
		9 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 9 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 9 } }
			}
		}
		8 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 8 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 8 } }
			}
		}
		7 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 7 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 7 } }
			}
		}
		6 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 6 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 6 } }
			}
		}
		5 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 5 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 5 } }
			}
		}
		4 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 4 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 4 } }
			}
		}
		3 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 3 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 3 } }
			}
		}
		2 = {
			if = {
				limit = {
					owner = { check_variable = { which = yuzu_$LEADER_CLASS$_skill value < 2 } }
				}
				owner = { set_variable = { which = yuzu_$LEADER_CLASS$_skill value = 2 } }
			}
		}
	}

	owner = {
		every_owned_leader = {
			limit = {
				leader_class = $LEADER_CLASS$
			}
			set_yuzu_leader_skill_single = { LEADER_CLASS = $LEADER_CLASS$ }
		}
		every_pool_leader = {
			limit = {
				leader_class = $LEADER_CLASS$
			}
			set_yuzu_leader_skill_single = { LEADER_CLASS = $LEADER_CLASS$ }
		}
	}
}
