building_yuzu_clone_vats = {
	icon = building_ancient_relics
	base_buildtime = @b1_time

	category = pop_assembly

	position_priority = 50

	planet_limit = {
		base = 1
	}

	potential = {
		exists = owner
		is_yuzu_empire = yes
		NOT = { has_modifier = resort_colony }
		OR = {
			owner = { is_ai = no }
			NAND = {
				free_district_slots = 0
				free_building_slots <= 1
				free_housing <= 0
				free_jobs <= 0
			}
		}
	}

	destroy_trigger = {
		exists = owner
		OR = {
			AND = {
				owner = { is_ai = yes }
				free_district_slots = 0
				free_building_slots = 0
				free_housing <= 0
				free_jobs <= 0
			}
			is_yuzu_empire = no
		}
	}

	triggered_planet_pop_group_modifier_for_species = {
		potential = { has_trait = trait_yuzu }
		bonus_pop_growth = 3
	}
	triggered_planet_modifier = {
		potential = { exists = owner }
		fake_pop_growth_mod = 3
	}
	triggered_planet_pop_group_modifier_for_species = {
		potential = {
			has_trait = trait_yuzu
			owner = { has_technology = tech_yuzu_rapid_cell_division_1 }
		}
		bonus_pop_growth = 1
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_technology = tech_yuzu_rapid_cell_division_1 }
		}
		fake_pop_growth_mod = 1
	}
	triggered_planet_pop_group_modifier_for_species = {
		potential = {
			has_trait = trait_yuzu
			owner = { has_technology = tech_yuzu_rapid_cell_division_2 }
		}
		bonus_pop_growth = 1
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_technology = tech_yuzu_rapid_cell_division_2 }
		}
		fake_pop_growth_mod = 1
	}
	triggered_planet_pop_group_modifier_for_species = {
		potential = {
			has_trait = trait_yuzu
			owner = { has_edict = yuzu_overloaded_clone }
		}
		bonus_pop_growth = 1
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_edict = yuzu_overloaded_clone }
		}
		fake_pop_growth_mod = 1
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_ascension_perk = ap_personality_transplantation }
		}
		planet_stability_add = 5
	}

	resources = {
