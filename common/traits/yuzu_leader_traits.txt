##### 星幽共鸣:舰队司令 #####
# 只给「指挥官 → 舰队司令」(subclass_commander_admiral) 的老练领袖选。
leader_trait_yuzu_astral_destiny_resonance_commander_admiral = {
	leader_trait_type = destiny
	inline_script = {
		script = trait/yuzu_destiny_icon
		CLASS = commander
		ICON = GFX_leader_trait_yuzu_astral_destiny_resonance
		RARITY = paragon
		COUNCIL = no
		TIER = none
	}
	background_icon = GFX_leader_background_destiny_1

	requires_traits = { trait_yuzu_astral }
	leader_class = { commander }
	leader_potential_add = {
		has_trait = subclass_commander_admiral
		has_trait = leader_trait_yuzu_astral
	}
	# 命定特质靠"升级时的候选列表"给出, 不能像普通特质那样写 randomized = no, 否则永远不会出现
	selectable_weight = {
		weight = @subclass_trait_weight
	}

	# 星幽术凝成舰队级的护盾与感知
	fleet_modifier = {
		ship_weapon_damage = 0.25
		ship_fire_rate_mult = 0.25
		ship_weapon_range_mult = 0.25
		ship_speed_mult = 0.25
		ships_upkeep_mult = -0.25
	}
	ai_weight = {
		weight = 100
	}
}
