# 培养
tr_yuzu_develop_adopt = {
	modifier = {
		leaders_upkeep_mult = -0.25
	}
}

tr_yuzu_develop_finish = {
	modifier = {
		species_leader_exp_gain = 0.5
		ascension_perks_add = 1
	}
}

# 突破上限
tr_yuzu_develop_1 = {
	custom_tooltip = tr_yuzu_develop_1_desc

	on_enabled = {
		hidden_effect = {
			every_owned_leader = {
				if = {
					limit = {
						has_base_skill >= 10
						NOR = {
							has_trait = leader_trait_yuzu_develop_scientist
							has_trait = leader_trait_yuzu_develop_commander
							has_trait = leader_trait_yuzu_develop_official
						}
					}
					if = {
						limit = {
							leader_class = scientist
						}
						add_trait = leader_trait_yuzu_develop_scientist
					}
					if = {
						limit = {
							leader_class = commander
						}
						add_trait = leader_trait_yuzu_develop_commander
					}
					if = {
						limit = {
							leader_class = official
						}
						add_trait = leader_trait_yuzu_develop_official
					}
				}
			}
			every_pool_leader = {
				if = {
					limit = {
						has_base_skill >= 10
						NOR = {
							has_trait = leader_trait_yuzu_develop_scientist
							has_trait = leader_trait_yuzu_develop_commander
							has_trait = leader_trait_yuzu_develop_official
						}
					}
					if = {
						limit = {
							leader_class = scientist
						}
						add_trait = leader_trait_yuzu_develop_scientist
					}
					if = {
						limit = {
							leader_class = commander
						}
						add_trait = leader_trait_yuzu_develop_commander
					}
					if = {
						limit = {
							leader_class = official
						}
						add_trait = leader_trait_yuzu_develop_official
					}
				}
			}
		}
	}

	ai_weight = {
		factor = 10000
	}
}
